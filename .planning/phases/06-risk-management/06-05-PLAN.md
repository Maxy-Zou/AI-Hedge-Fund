    ```python
    # ---------------------------------------------------------------------------
    # Phase 6 risk-manager node: deterministic-first veto with LLM advisory
    # rationale. Pattern 2 from 06-RESEARCH.md. First-violation-wins ordering:
    # position_size → sector → correlation → drawdown. LLM is CALLED even on
    # violations so rationale is populated, but node OVERWRITES LLM status via
    # model_copy(update={...}) — the LLM CANNOT decide the outcome (T-06-02).
    # ---------------------------------------------------------------------------


    async def risk_manager_node(state: DebatePipelineState, deps: RiskDeps) -> dict:
        """Phase-6 Risk Manager node — deterministic checks + advisory rationale.

        Threat mitigations:
            T-06-02 (veto bypass): status is set via model_copy AFTER agent runs.
            T-06-04 (policy drift): policy_sha persisted on every RiskAssessment.
            T-06-05 (div-by-zero): risk/drawdown.py guards handle NaN paths.
            Pitfall 1: LLM rationale is advisory; deterministic result wins.
            Pitfall 2: policy + portfolio loaded INSIDE the node.
            Pitfall 7: router fails closed on missing assessment.
        """
        if state.get("error"):
            return {}
        thesis = state.get("thesis")
        if thesis is None:
            logger.error("risk_no_thesis", ticker=state.get("ticker"))
            return {"error": "No thesis available for risk check"}
        # ThesisOutput schema (src/ai_hedge_fund/schemas/agents.py line 82) has:
        #   ticker, bull_case, bear_case, confidence: int (0-100), risk_factors.
        # NO sector or instrument_type — those come from state["candidate_metadata"].
        confidence = thesis.get("confidence")
        if confidence is None:
            logger.error("risk_thesis_missing_confidence", ticker=state.get("ticker"))
            return {"error": "Thesis missing 'confidence' field"}
        if confidence >= 75:
            conviction = "high"
        elif confidence >= 50:
            conviction = "medium"
        else:
            conviction = "low"

        # Load policy + portfolio INSIDE the node (Pitfall 2).
        policy = load_policy(deps.policy_path) if deps.policy_path else load_policy()
        policy_sha = compute_policy_sha(policy)
        portfolio = load_portfolio(deps.db_session, state["as_of_date"])

        candidate_ticker = thesis.get("ticker") or state["ticker"]
        # Candidate sector / instrument_type come from state["candidate_metadata"] — NOT
        # ThesisOutput (which has no sector field). If metadata is missing, sector is
        # "Unknown" (safe default: never matches excluded_sectors, never matches a
        # known sector-concentration threshold).
        candidate_meta = state.get("candidate_metadata") or {}
        candidate_sector = candidate_meta.get("sector") or "Unknown"
        candidate_instrument_type = candidate_meta.get("instrument_type") or "equity"
        candidate_size_pct = derive_candidate_size_pct(conviction, policy)

        # Run four deterministic checks in fixed order; first violation wins.
        violation: Violation | None = (
            check_exclusions(candidate_sector, candidate_instrument_type, policy)
            or check_position_size(candidate_size_pct, policy)
            or check_sector_concentration(candidate_sector, candidate_size_pct, portfolio, policy)
            or check_correlation(candidate_ticker, portfolio, deps.returns, policy)
            or check_drawdown(candidate_ticker, candidate_size_pct, portfolio, deps.returns, policy)
        )
        status = "VETOED" if violation else "APPROVED"

        # Call LLM for rationale (even on veto — audit trail).
        prompt = format_risk_context_for_rationale(
            thesis=thesis, status=status, violation=violation,
            portfolio=portfolio, policy=policy,
        )
        try:
            limits = get_risk_manager_limits()
            result = await risk_manager_agent.run(prompt, usage_limits=limits)
        except UsageLimitExceeded as e:
            logger.error("risk_manager_budget_exceeded", ticker=state["ticker"], error=str(e))
            return {"error": f"Risk manager budget exceeded: {e}"}

        # DETERMINISTIC OVERWRITE — LLM status value is DISCARDED (Pitfall 1).
        assessment = RiskAssessment(
            ticker=candidate_ticker,
            status=status,  # Python-authored, not LLM-authored
            constraint_violated=violation.name if violation else None,
            observed=violation.observed if violation else None,
            limit=violation.limit if violation else None,
            rationale=result.output.rationale,  # LLM produces rationale only
            policy_sha=policy_sha,
        )

        usage = result.usage()
        event_name = "risk_manager_complete" if status == "APPROVED" else "risk_manager_veto"
        logger.info(
            event_name,
            ticker=state["ticker"],
            status=status,
            constraint_violated=assessment.constraint_violated,
            observed=assessment.observed,
            limit=assessment.limit,
            policy_sha=policy_sha,
            input_tokens=usage.input_tokens,
            output_tokens=usage.output_tokens,
            total_tokens=usage.total_tokens,
        )
        return {"risk_assessment": assessment.model_dump()}


    def route_after_risk(state: DebatePipelineState) -> Literal["signal", "__end__"]:
        """Fail-closed router: APPROVED → signal; VETOED or missing → END.

        Pitfall 7 mitigation: missing/None risk_assessment routes to __end__, not signal.
        """
        assessment = state.get("risk_assessment")
        if assessment is None:
            return "__end__"
        return "signal" if assessment.get("status") == "APPROVED" else "__end__"
    ```
