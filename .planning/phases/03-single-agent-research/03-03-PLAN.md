```python
class ThesisPoint(BaseModel):
    claim: str = Field(min_length=1)
    evidence: str = Field(min_length=1)
    source_tool: str = Field(min_length=1)

class ThesisOutput(BaseModel):
    ticker: str
    bull_case: list[ThesisPoint] = Field(min_length=3)
    bear_case: list[ThesisPoint] = Field(min_length=3)
    confidence: int = Field(ge=0, le=100)
    risk_factors: list[str] = Field(min_length=2)

class SignalOutput(BaseModel):
    ticker: str
    direction: Literal["long", "short", "neutral"]
    conviction: Literal["low", "medium", "high"]
    time_horizon: str
    position_size_pct: float = Field(ge=0.0, le=100.0)
    thesis_summary: str
```
