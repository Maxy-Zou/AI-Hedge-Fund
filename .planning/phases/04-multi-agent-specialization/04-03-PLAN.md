import operator
from typing import Annotated, Required
from typing_extensions import TypedDict

class MultiAgentPipelineState(TypedDict, total=False):
    """State flowing through the Phase-4 multi-agent pipeline.

    Uses Annotated[list, operator.add] reducer on analyst_reports so each
    parallel analyst node can append its report without overwriting others.
    LangGraph's built-in reducer concatenates the lists from all parallel
    nodes.

    Required fields:
        ticker: Stock ticker symbol being analyzed.
        as_of_date: Temporal cutoff as ISO-format date string.

    Optional fields:
        analyst_reports: Accumulated analyst outputs (appended via reducer).
        thesis: ThesisOutput.model_dump() from manager_node.
        signal: SignalOutput.model_dump() from signal_node.
        error: Error message if manager or signal node failed.
    """

    ticker: Required[str]
    as_of_date: Required[str]
    analyst_reports: Annotated[list[dict], operator.add]
    thesis: dict | None
    signal: dict | None
    error: str | None
```

Note: `operator` and `Annotated` must be imported at the top of the file. `Required` and `TypedDict` are already imported.

from ai_hedge_fund.graph.nodes import (
    # existing imports stay...
    fundamental_node,
    sentiment_node,
    technical_node,
    manager_node,
    multi_agent_signal_node,
)
from ai_hedge_fund.schemas.state import (
    PipelineState,
    ResearchPipelineState,
    MultiAgentPipelineState,
)

def build_multi_agent_pipeline(
    checkpointer: BaseCheckpointSaver | None = None,
) -> CompiledStateGraph:
    """Build the multi-agent pipeline: analysts (parallel) -> manager -> signal.

    Phase-4 pipeline that takes a ticker + as_of_date and runs three
    specialist analysts in parallel (Fundamental, Sentiment, Technical),
    synthesizes their reports through a Research Manager, then generates
    a trade signal. Uses LangGraph static fan-out (START -> 3 analysts)
    and fan-in (3 analysts -> manager) with Annotated[list, operator.add]
    reducer on analyst_reports to merge parallel outputs.

    Graph topology is compile-time static (threat model T-04-09).
    Every node run is bounded by UsageLimits (T-04-04, T-04-07).

    Args:
        checkpointer: Optional checkpoint saver for state persistence.

    Returns:
        Compiled StateGraph ready for invocation via ainvoke.
    """
    builder = StateGraph(MultiAgentPipelineState)

    # Add nodes
    builder.add_node("fundamental", fundamental_node)
    builder.add_node("sentiment", sentiment_node)
    builder.add_node("technical", technical_node)
    builder.add_node("manager", manager_node)
    builder.add_node("signal", multi_agent_signal_node)

    # Fan-out: START -> all 3 analysts in parallel
    builder.add_edge(START, "fundamental")
    builder.add_edge(START, "sentiment")
    builder.add_edge(START, "technical")

    # Fan-in: all 3 analysts -> manager
    builder.add_edge("fundamental", "manager")
    builder.add_edge("sentiment", "manager")
    builder.add_edge("technical", "manager")

    # Manager -> signal -> END
    builder.add_edge("manager", "signal")
    builder.add_edge("signal", END)

    return builder.compile(checkpointer=checkpointer)
```
