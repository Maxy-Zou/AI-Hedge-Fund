       - from __future__ import annotations
       - Import Agent from pydantic_ai, UsageLimits from pydantic_ai.usage, BaseModel from pydantic
       - Import ModelTier, AgentBudget, MODEL_BUDGETS from ai_hedge_fund.models

       - class BudgetExceededError(Exception): pass
         Store agent_name (str) and tokens_used (int) and tokens_limit (int) as attributes.

       - def create_agent(model_tier: ModelTier, output_type: type[BaseModel], system_prompt: str, **kwargs) -> Agent:
         Creates and returns Agent(model_tier.value, output_type=output_type, system_prompt=system_prompt, **kwargs).
         This is a thin factory -- the value is centralizing model string resolution and ensuring every agent goes through a single creation path.

       - def get_usage_limits(model_tier: ModelTier, *, input_override: int | None = None, output_override: int | None = None, total_override: int | None = None) -> UsageLimits:
         Looks up MODEL_BUDGETS[model_tier], creates UsageLimits(input_tokens_limit=input_override or budget.input_tokens_limit, output_tokens_limit=output_override or budget.output_tokens_limit, total_tokens_limit=total_override or budget.total_tokens_limit).

       - @dataclass(frozen=True) class AgentUsageRecord:
           agent_name: str
           input_tokens: int
           output_tokens: int
           total_tokens: int

       - @dataclass(frozen=True) class PipelineBudgetTracker:
           total_limit: int (default 500_000)
           records: tuple[AgentUsageRecord, ...] (default empty tuple -- immutable sequence)

           @property total_used -> int: sum of record.total_tokens for all records
           @property remaining_budget -> int: total_limit - total_used
           def record_usage(self, agent_name: str, input_tokens: int, output_tokens: int, total_tokens: int) -> PipelineBudgetTracker:
             new_record = AgentUsageRecord(agent_name, input_tokens, output_tokens, total_tokens)
             new_records = self.records + (new_record,)
             new_tracker = PipelineBudgetTracker(total_limit=self.total_limit, records=new_records)
             if new_tracker.total_used > self.total_limit:
               raise BudgetExceededError(agent_name, new_tracker.total_used, self.total_limit)
             return new_tracker
           def usage_summary(self) -> dict: returns {"total_used": self.total_used, "total_limit": self.total_limit, "remaining": self.remaining_budget, "breakdown_by_agent": {r.agent_name: r.total_tokens for r in self.records}}

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| LLM response -> PydanticAI output schema | Untrusted LLM output validated against Pydantic BaseModel constraints |
| Agent factory -> model tier | Model routing centralized through enum to prevent arbitrary model selection |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-02-01 | Denial of Service | agents/base.py | mitigate | UsageLimits enforced on every agent.run() call; PipelineBudgetTracker enforces pipeline-level cap; BudgetExceededError halts execution |
| T-02-02 | Tampering | agents/base.py | mitigate | AgentBudget and PipelineBudgetTracker are frozen dataclasses -- budget values cannot be modified after creation |
| T-02-03 | Repudiation | agents/base.py | mitigate | AgentUsageRecord captures per-agent token usage; usage_summary provides audit trail of all agent costs |
</threat_model>
