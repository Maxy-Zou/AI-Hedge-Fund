    ```python
    Every belief YAML under beliefs/ validates against :class:`Belief`.
    Unknown keys raise ValidationError (Pitfall 4 -- typo 'confidance'

    event, once appended, is history and must not mutate in place.

    Belief is deliberately NOT frozen: the writer in
    ``ai_hedge_fund.memory.beliefs.write_belief`` produces a new
    raw-dict for atomic file replacement; Pydantic immutability would
    force unnecessary re-validation churn on each patch. The
    human-edit contract (MEM-03) is enforced by the writer, not by
    pydantic.
    ```

    ```python
    """Belief memory I/O -- ruamel.yaml round-trip + human-edit guard.

    This module is the SOLE mutation entry point for belief YAML files
    (MEM-03 contract). All machine writers -- Plan 07-04 self-critique,
    any future agent-authored belief update -- MUST call write_belief.
    Nothing else in the system is permitted to touch a belief file.

    Threat mitigations:
        T-07-10 (YAML RCE): ruamel.yaml.YAML() is safe by default
                (does NOT instantiate arbitrary Python). Plain
                yaml.load(data) is forbidden; this module imports
                ruamel only.
        T-07-11 (Path traversal): belief_path_for_ticker enforces a
                regex guard (``[A-Z0-9.\\-]{1,10}``) BEFORE joining
                ticker into a filesystem path.
        T-07-12 (Silent human-edit overwrite): write_belief is the
                single writer; skip-if-human-edited + field_locks
                audit every patched field. Returned dict records
                applied vs skipped with reason codes.
        T-07-13 (Partial-write corruption): atomic tmp+rename
                (Path.replace is atomic on POSIX) means a mid-write
                crash leaves the original file intact.
    """

    from __future__ import annotations

    import re
    from pathlib import Path
    from typing import Any

    import structlog
    from ruamel.yaml import YAML

    from ai_hedge_fund.schemas.memory import Belief

    logger = structlog.get_logger(__name__)

    _TICKER_RE = re.compile(r"^[A-Z0-9.\-]{1,10}$")
    _OVERRIDE_META_FIELDS = frozenset({"human_edited", "edited_at", "field_locks"})
    _HUMAN_GUARDED_FIELDS = frozenset({"thesis", "confidence"})


    def _yaml() -> YAML:
        """Return a YAML() instance configured for round-trip safety."""
        y = YAML()
        y.preserve_quotes = True
        return y


    def belief_path_for_ticker(beliefs_dir: Path, ticker: str) -> Path:
        """Return beliefs_dir/tickers/<ticker>.yaml with a regex guard.

        Pitfall 9: rejects path-traversal attempts (``"../foo"``),
        lowercase tickers, and anything longer than 10 chars. Same
        regex the SEC ticker fields use elsewhere in the project.
        """
        if not _TICKER_RE.fullmatch(ticker):
            raise ValueError(f"Invalid ticker {ticker!r} (expected [A-Z0-9.\\-]{{1,10}})")
        return beliefs_dir / "tickers" / f"{ticker}.yaml"


    def load_belief(path: Path) -> tuple[Belief, Any]:
        """Load and validate a belief YAML file.

        Returns:
            Tuple ``(Belief, raw)``. ``raw`` is the round-trip ruamel
            object suitable for ``write_belief``; write operations MUST
            be applied to this object (not to ``Belief.model_dump()``)
            so comments and key order survive.

        Raises:
            FileNotFoundError: File does not exist.
            pydantic.ValidationError: YAML is not a valid Belief.
        """
        yaml = _yaml()
        raw = yaml.load(path)
        if raw is None:
            raise ValueError(f"Belief file {path} is empty")
        belief = Belief.model_validate(dict(raw))
        return belief, raw


    def write_belief(
        path: Path,
        raw: Any,
        patches: dict[str, Any],
        *,
        skip_if_human_edited: bool = True,
    ) -> dict[str, Any]:
        """Apply patches to ``raw`` and atomically rewrite ``path``.

        Override-meta fields (``human_edited``, ``edited_at``,
        ``field_locks``) are NEVER written by this function. Per-field
        locks in ``raw['field_locks']`` veto the corresponding patch.
        When ``raw['human_edited']`` is true and ``skip_if_human_edited``
        is true, the global guard vetoes ``thesis`` and ``confidence``
        patches regardless of ``field_locks``.

        The file is rewritten via tmp+rename so a crash mid-write does
        not corrupt the original.

        Args:
            path: Target YAML file (same file ``load_belief`` read).
            raw: Round-trip object from ``load_belief``.
            patches: ``{field: new_value}`` map of proposed writes.
            skip_if_human_edited: When true (default), block thesis and
                confidence patches when the YAML has ``human_edited:
                true``. Set false only in offline repair scripts where
                an operator has explicitly asked to override.

        Returns:
            ``{"applied": [..], "skipped": {field: reason, ...}}`` audit
            dict. Reasons are one of: ``writer_never_touches_override_meta``,
            ``field_locked_by_human``, ``human_edited_global_flag_set``.
            Version is incremented iff at least one field was applied.
        """
        applied: list[str] = []
        skipped: dict[str, str] = {}
        locks = raw.get("field_locks", {}) or {}
        human_edited = bool(raw.get("human_edited"))

        for field, value in patches.items():
            if field in _OVERRIDE_META_FIELDS:
                skipped[field] = "writer_never_touches_override_meta"
                continue
            if locks.get(field) is True:
                skipped[field] = "field_locked_by_human"
                continue
            if (
                human_edited
                and skip_if_human_edited
                and field in _HUMAN_GUARDED_FIELDS
            ):
                skipped[field] = "human_edited_global_flag_set"
                continue
            raw[field] = value
            applied.append(field)

        if applied:
            raw["version"] = int(raw.get("version", 0)) + 1

        yaml = _yaml()
        tmp = path.with_suffix(path.suffix + ".tmp")
        try:
            with tmp.open("w") as fh:
                yaml.dump(raw, fh)
            tmp.replace(path)  # atomic on POSIX
        except Exception:
            # Clean up the tmp if the rename didn't happen.
            if tmp.exists():
                tmp.unlink(missing_ok=True)
            raise

        logger.info(
            "belief_write",
            path=str(path),
            applied=applied,
            skipped=skipped,
            new_version=raw.get("version"),
        )
    ```
