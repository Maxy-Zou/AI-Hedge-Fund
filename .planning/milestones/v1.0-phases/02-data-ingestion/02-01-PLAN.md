       a. `SecFiling(Base, DualTimestampMixin)`:
          - __tablename__ = "sec_filings"
          - id: Mapped[int] = mapped_column(primary_key=True)
          - ticker: Mapped[str] = mapped_column(String(10), nullable=False, index=True)
          - accession_no: Mapped[str] = mapped_column(String(30), nullable=False)
          - form_type: Mapped[str] = mapped_column(String(10), nullable=False)  # 10-K, 10-Q, 8-K
          - filing_date: Mapped[datetime] = mapped_column(DateTime(timezone=True), nullable=False)
          - sections_json: Mapped[str | None] = mapped_column(Text, nullable=True)  # JSON-serialized section text
          - summary_text: Mapped[str | None] = mapped_column(Text, nullable=True)  # NL summary
          - UniqueConstraint("ticker", "accession_no", name="uq_sec_filings_ticker_accession")

       b. `XbrlFact(Base, DualTimestampMixin)`:
          - __tablename__ = "xbrl_facts"
          - id, ticker(String(10), indexed), cik(String(10)), concept(String(100)), value_cents(BigInteger),
            unit(String(20)), fiscal_period(String(10)), fiscal_year(Integer), filed_date(DateTime(tz=True))
          - UniqueConstraint("ticker", "concept", "fiscal_period", "fiscal_year", name="uq_xbrl_facts_ticker_concept_period_year")

       c. `DailyPrice(Base, DualTimestampMixin)`:
          - __tablename__ = "daily_prices"
          - id, ticker(String(10), indexed), trade_date(Date, not null), open_cents(BigInteger),
            high_cents(BigInteger), low_cents(BigInteger), close_cents(BigInteger),
            adj_close_cents(BigInteger), volume(BigInteger), source(String(20), default="yfinance")
          - UniqueConstraint("ticker", "trade_date", "source", name="uq_daily_prices_ticker_date_source")

       d. `InsiderTrade(Base, DualTimestampMixin)`:
          - __tablename__ = "insider_trades"
          - id, ticker(String(10), indexed), insider_name(String(200)), insider_title(String(200), nullable=True),
            trade_type(String(20)), shares(BigInteger), price_cents(BigInteger, nullable=True),
            value_cents(BigInteger, nullable=True), trade_date(Date), filing_date(DateTime(tz=True))
          - UniqueConstraint("ticker", "insider_name", "trade_date", "trade_type", "shares", name="uq_insider_trades_dedup")

       e. `NewsArticle(Base, DualTimestampMixin)`:
          - __tablename__ = "news_articles"
          - id, ticker(String(10), indexed), headline(String(500)), source(String(100)),
            url(String(1000)), sentiment_score(Float, nullable=True),
            published_date(DateTime(tz=True))
          - UniqueConstraint("ticker", "url", name="uq_news_articles_ticker_url")

       f. `MacroIndicator(Base, DualTimestampMixin)`:
          - __tablename__ = "macro_indicators"
          - id, series_id(String(20), indexed), series_name(String(200)),
            value(Float), observation_date(Date)
          - UniqueConstraint("series_id", "observation_date", name="uq_macro_indicators_series_date")
