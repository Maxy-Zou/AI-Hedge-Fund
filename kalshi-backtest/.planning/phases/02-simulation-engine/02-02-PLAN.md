    ```python
    def calculate_fee_cents(contracts: int, price_fraction: float) -> int:
        """Kalshi taker fee in integer cents. Formula: ceil(0.07 * C * P * (1-P)).
        Price_fraction must be in [0.0, 1.0], NOT cents. Raises ValueError if out of range."""
        if contracts <= 0:
            raise ValueError(f"contracts must be >= 1, got {contracts}")
        if not 0.0 <= price_fraction <= 1.0:
            raise ValueError(f"price_fraction must be in [0.0, 1.0], got {price_fraction}")
        raw = 0.07 * contracts * price_fraction * (1.0 - price_fraction)
        return max(1, math.ceil(raw))

    def simulate_fill_price(
        direction: str,
        limit_price: int,
        mid_price: int,
        spread_floor: int = 1,
    ) -> int | None:
        """Fill price in cents, None if limit is not fillable.
        YES buy: fills at mid + ceil(spread_floor/2); must be <= limit_price.
        NO buy: fills at mid - ceil(spread_floor/2); 100 - fill must be <= limit_price."""

    def calculate_settlement_pnl(
        direction: str,
        entry_price: int,
        contracts: int,
        result: str,
    ) -> int:
        """Hold-to-settlement P&L in cents."""

    def calculate_exit_pnl(
        direction: str,
        entry_price: int,
        exit_price: int,
        contracts: int,
    ) -> int:
        """Mark-to-market exit P&L in cents."""
    ```
