    3. Create `src/ai_washer/universe/types.py` with these Pydantic models:

       ```python
       from __future__ import annotations
       from pydantic import BaseModel, Field, model_validator

       class EFTSHit(BaseModel):
           """Single EFTS search result."""
           accession_no: str
           form_type: str
           file_date: str
           entity_name: str
           ciks: list[str]
           period_of_report: str | None = None
           display_names: list[str] = Field(default_factory=list)

       class EFTSResponse(BaseModel):
           """Parsed EFTS API response."""
           total_value: int
           total_relation: str  # "eq" or "gte"
           hits: list[EFTSHit]

           @property
           def is_truncated(self) -> bool:
               return self.total_relation == "gte"

       class MarketCapRange(BaseModel):
           min_cents: int = Field(gt=0)
           max_cents: int = Field(gt=0)

           @model_validator(mode="after")
           def _check_min_lt_max(self) -> MarketCapRange:
               if self.min_cents >= self.max_cents:
                   raise ValueError(f"min_cents ({self.min_cents}) must be less than max_cents ({self.max_cents})")
               return self
       ```
