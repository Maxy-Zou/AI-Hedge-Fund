```python
class FilingCollector:
    """Orchestrates SEC filing retrieval and XBRL extraction for target companies.

    Coordinates FilingClient (filing text) and XBRLExtractor (financial data),
    persisting results to the database with idempotent daily checks.
    """
```

Key methods:

1. `__init__(self, app_settings: AppSettings, filing_settings: FilingCollectionSettings | None = None)`:
   - Store settings
   - Create session factory via `create_engine_from_settings(app_settings)` and `get_session_factory(engine)`
   - Initialize structlog logger

2. `collect_for_company(self, company_id: uuid.UUID, cik: str, ticker: str, collection_date: date | None = None) -> CollectionResult`:
   - Default collection_date to `date.today()` (UTC)
   - Create FilingClient and XBRLExtractor as context managers using `app_settings.edgar_identity`
   - **Filing collection loop** (for each form_type in filing_settings.form_types):
     - Call `filing_client.get_latest_filings(form_type, cik)`
     - For each returned FilingData:
       - **Idempotency check**: query `session.execute(select(Filing).where(Filing.company_id == company_id, Filing.form_type == form_type, Filing.accession_no == filing_data.accession_no))` -- skip if exists, increment skipped_count
       - Create Filing ORM object:
         - `company_id=company_id`
         - `form_type=filing_data.form_type`
         - `accession_no=filing_data.accession_no`
         - `filing_date=filing_data.filing_date`
         - `period_of_report=filing_data.period_of_report`
         - `sections=filing_data.sections.model_dump()` (JSONB)
         - `content_hash=filing_data.content_hash`
         - `collection_metadata={"collector_version": "0.3.0", "collected_at": datetime.utcnow().isoformat()}`
         - `as_of_date=filing_data.period_of_report or filing_data.filing_date` (per Pitfall 4: period_of_report is the business date)
         - `observed_date=collection_date` (when we collected it)
       - `session.add(filing)`, increment filing_count
     - Catch exceptions per form_type -- add to errors list, continue to next form_type
   - **XBRL extraction**:
     - Call `xbrl_extractor.extract_company_xbrl(cik)`
     - For each XBRLFactRecord:
       - **Idempotency check**: query `select(XBRLFact).where(XBRLFact.company_id == company_id, XBRLFact.concept == fact.concept, XBRLFact.end_date == fact.end_date, XBRLFact.fiscal_period == fact.fiscal_period)` -- skip if exists
       - Create XBRLFact ORM object with all fields mapped from XBRLFactRecord
       - `as_of_date=fact.end_date` (the fiscal period this fact represents)
       - `observed_date=collection_date`
       - `session.add(xbrl_fact)`, increment xbrl_fact_count
     - Catch XBRL extraction errors -- add to errors list
   - `session.commit()`
   - Return CollectionResult

3. `collect_all(self, collection_date: date | None = None) -> list[CollectionResult]`:
   - Default collection_date to `date.today()`
   - Query all active companies: `select(Company).where(Company.is_active == True, Company.cik.isnot(None))`
   - For each company, call `collect_for_company(company.id, company.cik, company.ticker, collection_date)`
   - Sleep `_SEC_RATE_LIMIT_DELAY` between companies (from edgar_client module)
   - Log progress every 10 companies
   - Return list of CollectionResult

```python
@collect_app.command(name="company")
def collect_company(
    identifier: str = typer.Argument(help="Company ticker or CIK to collect filings for"),
    dry_run: bool = typer.Option(False, "--dry-run", help="Show what would happen without writing to DB"),
):
    """Collect SEC filings and XBRL data for a single company."""
    from sqlalchemy import select
    from ai_washer.config import FilingCollectionSettings, load_app_settings
    from ai_washer.db.models import Company
    from ai_washer.db.session import create_engine_from_settings, get_session_factory
    from ai_washer.ingestion.filing_collector import FilingCollector

    app_settings = load_app_settings()
    # Look up company by ticker or CIK
    engine = create_engine_from_settings(app_settings)
    session_factory = get_session_factory(engine)
    with session_factory() as session:
        stmt = select(Company).where(
            (Company.ticker == identifier.upper()) | (Company.cik == identifier)
        )
        company = session.execute(stmt).scalar_one_or_none()
        if company is None:
            typer.echo(f"Company not found: {identifier}", err=True)
            raise typer.Exit(code=1)
        if company.cik is None:
            typer.echo(f"Company {company.ticker} has no CIK -- cannot collect filings", err=True)
            raise typer.Exit(code=1)

    if dry_run:
        typer.echo(f"DRY RUN: Would collect filings for {company.ticker} (CIK: {company.cik})")
        typer.echo("Filing types: 10-K, 10-Q, 8-K")
        typer.echo("XBRL concepts: rd_expense, capex, revenue")
        return

    collector = FilingCollector(app_settings=app_settings)
    result = collector.collect_for_company(
        company_id=company.id, cik=company.cik, ticker=company.ticker,
    )
    typer.echo(f"Company: {company.ticker} (CIK: {company.cik})")
    typer.echo(f"Filings collected: {result.filing_count}")
    typer.echo(f"XBRL facts extracted: {result.xbrl_fact_count}")
    typer.echo(f"Skipped (already collected): {result.skipped_count}")
    if result.errors:
        typer.echo(f"Errors: {len(result.errors)}")
        for err in result.errors:
            typer.echo(f"  - {err}", err=True)


@collect_app.command(name="all")
def collect_all(
    dry_run: bool = typer.Option(False, "--dry-run", help="Show what would happen without writing to DB"),
):
    """Collect SEC filings and XBRL data for all active companies in universe."""
    from ai_washer.config import load_app_settings
    from ai_washer.ingestion.filing_collector import FilingCollector

    app_settings = load_app_settings()

    if dry_run:
        # Count active companies with CIK
        from sqlalchemy import func, select
        from ai_washer.db.models import Company
        from ai_washer.db.session import create_engine_from_settings, get_session_factory
        engine = create_engine_from_settings(app_settings)
        sf = get_session_factory(engine)
        with sf() as session:
            count = session.execute(
                select(func.count()).where(Company.is_active == True, Company.cik.isnot(None))
            ).scalar()
        typer.echo(f"DRY RUN: Would collect filings for {count} active companies")
        return

    collector = FilingCollector(app_settings=app_settings)
    results = collector.collect_all()
    total_filings = sum(r.filing_count for r in results)
    total_xbrl = sum(r.xbrl_fact_count for r in results)
    total_skipped = sum(r.skipped_count for r in results)
    total_errors = sum(len(r.errors) for r in results)
    typer.echo(f"Collection complete for {len(results)} companies")
    typer.echo(f"Total filings: {total_filings}")
    typer.echo(f"Total XBRL facts: {total_xbrl}")
    typer.echo(f"Total skipped: {total_skipped}")
    if total_errors:
        typer.echo(f"Total errors: {total_errors}")
```
